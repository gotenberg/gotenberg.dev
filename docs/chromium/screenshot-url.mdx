Captures any page Chromium can reach as a PNG, JPEG or WebP image.

<Endpoint path="/forms/chromium/screenshot/url" />

## Capture a page

<Request
  route="/forms/chromium/screenshot/url"
  fields={{ url: "https://example.com" }}
  output="example.png"
/>

<Screenshot
  route="/forms/chromium/screenshot/url"
  fields={{ url: "https://example.com" }}
  name="example"
  selector="p"
/>

<Url route="/forms/chromium/screenshot/url" output="example.png" />
