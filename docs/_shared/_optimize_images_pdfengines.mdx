import ApiEndpoint from "@site/src/components/documentation/ApiEndpoint";

Reduces the resulting file size by re-encoding its images to JPEG during post-processing. Text, vectors, fonts, and structure are left untouched.

{props.showCurlExample && (

<ApiEndpoint
  formFields={[
    {
      name: "optimizeImages",
      type: "boolean",
      defaultValue: "false",
      description:
        "Re-encodes the images in the resulting PDF to reduce its file size.",
    },
    {
      name: "imageQuality",
      type: "integer",
      defaultValue: "80",
      description:
        "The JPEG quality applied to each re-encoded image, between 1 and 100. Only used if `optimizeImages` is `true`.",
    },
  ]}
  curl={`
curl \\
--request ${props.method} http://localhost:3000${props.path} \\
${props.curlFormData}
--form optimizeImages=true \\
--form imageQuality=80 \\
${props.curlOutput}
`}
/>
)}
